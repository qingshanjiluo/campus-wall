        // ============================================
        // 加载动画：序列式打字机效果
        // 模拟Galgame开场时的文本逐字浮现
        // ============================================
        const loadingTexts = [
            "正在打开社团活动室的门...",
            "阳光透过百叶窗洒了进来...",
            "欢迎来到校园墙 ✨"
        ];
        
        let textIndex = 0;
        let charIndex = 0;
        const loadingTextEl = document.getElementById('loadingText');
        const loadingScreen = document.getElementById('loadingScreen');
        const loadingLogo = document.querySelector('.loading-logo');

        // 显示Logo
        setTimeout(() => {
            loadingLogo.style.transition = 'all 0.8s ease';
            loadingLogo.style.opacity = '1';
            loadingLogo.style.transform = 'translateY(0)';
        }, 300);

        // 打字机效果
        function typeWriter() {
            if (textIndex < loadingTexts.length) {
                if (charIndex < loadingTexts[textIndex].length) {
                    loadingTextEl.innerHTML = loadingTexts[textIndex].substring(0, charIndex + 1) + '<span class="loading-cursor"></span>';
                    loadingTextEl.style.opacity = '1';
                    charIndex++;
                    setTimeout(typeWriter, 60);
                } else {
                    setTimeout(() => {
                        textIndex++;
                        charIndex = 0;
                        if (textIndex < loadingTexts.length) {
                            typeWriter();
                        } else {
                            // 加载完成，淡出
                            setTimeout(() => {
                                loadingScreen.classList.add('hidden');
                                initAnimations();
                            }, 800);
                        }
                    }, 600);
                }
            }
        }

        setTimeout(typeWriter, 1200);

        // ============================================
        // GSAP 滚动动画
        // 所有.reveal元素在滚动进入视口时触发淡入上浮
        // ============================================
        function initAnimations() {
            // 导航栏入场
            gsap.to('#navbar', {
                opacity: 1,
                y: 0,
                duration: 0.8,
                ease: 'power2.out'
            });

            // 注册ScrollTrigger
            gsap.registerPlugin(ScrollTrigger);

            // 为所有.reveal元素添加滚动触发
            gsap.utils.toArray('.reveal').forEach((elem, i) => {
                gsap.to(elem, {
                    scrollTrigger: {
                        trigger: elem,
                        start: 'top 85%',
                        toggleActions: 'play none none none'
                    },
                    opacity: 1,
                    y: 0,
                    duration: 0.8,
                    delay: i % 3 * 0.1, // 交错动画
                    ease: 'power2.out'
                });
            });

            // 背景光晕视差效果
            gsap.to('.bg-glow-1', {
                scrollTrigger: {
                    trigger: 'body',
                    start: 'top top',
                    end: 'bottom bottom',
                    scrub: 1
                },
                y: -100
            });

            gsap.to('.bg-glow-2', {
                scrollTrigger: {
                    trigger: 'body',
                    start: 'top top',
                    end: 'bottom bottom',
                    scrub: 1
                },
                y: -200
            });

            gsap.to('.bg-glow-3', {
                scrollTrigger: {
                    trigger: 'body',
                    start: 'top top',
                    end: 'bottom bottom',
                    scrub: 1
                },
                y: -150
            });
        }
